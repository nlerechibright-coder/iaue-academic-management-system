--- FILE: institution/views.py (faculty_create function) ---
def faculty_create(request):
    if not is_management(request.user): return redirect('login')
    if request.method == 'POST':
        name = request.POST.get('name')
        code = request.POST.get('code')
        if Faculty.objects.filter(code=code).exists():
            messages.error(request, "Faculty Code already exists.")
        else:
            Faculty.objects.create(name=name, code=code)
            messages.success(request, f"Faculty '{name}' created.")
            return redirect('institution:faculty_list')
    return render(request, 'management/faculty_form.html')

@login_required
